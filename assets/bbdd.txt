-- ============================================================
-- BASE DE DATOS: hundir_flota
-- ============================================================

DROP DATABASE IF EXISTS hundir_flota;
CREATE DATABASE hundir_flota CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE hundir_flota;

-- ============================================================
-- 1️⃣ Tabla: usuario
-- ============================================================
CREATE TABLE usuario (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombreUsuario VARCHAR(50) NOT NULL UNIQUE,
    pass VARCHAR(255) NOT NULL,
    imagenPerfil VARCHAR(255) DEFAULT 'assets/img/default-avatar.png',
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    victorias INT DEFAULT 0
) ENGINE=InnoDB;

-- ============================================================
-- 2️⃣ Tabla: partidas
-- ============================================================
CREATE TABLE partidas (
    idPartida INT AUTO_INCREMENT PRIMARY KEY,
    nombreUsuario VARCHAR(50) NOT NULL,
    nombreOponente VARCHAR(50) NOT NULL,
    ganador VARCHAR(50),
    puntos INT DEFAULT 0,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (nombreUsuario) REFERENCES usuario(nombreUsuario)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (nombreOponente) REFERENCES usuario(nombreUsuario)
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;

-- ============================================================
-- 3️⃣ Tabla: almirantes
-- ============================================================
CREATE TABLE almirantes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombreAlmirante VARCHAR(75) NOT NULL,
    imagenAlmirante VARCHAR(100),
    victorias INT DEFAULT 0
) ENGINE=InnoDB;

-- ============================================================
-- 7️⃣ Tabla: barcos
-- ============================================================
CREATE TABLE barcos (
    idBarco INT AUTO_INCREMENT PRIMARY KEY,
    idPartida INT NOT NULL,
    propietario ENUM('jugador','enemigo') NOT NULL,
    tipoBarco ENUM('portaaviones','acorazado','destructor','fragata','corbeta') NOT NULL,
    size INT NOT NULL,                -- total de casillas del barco
    ancho INT NOT NULL,               -- ancho del barco (para rectangulares)
    alto INT NOT NULL,                -- alto del barco
    orientacion ENUM('horizontal','vertical') NOT NULL,
    xInicio INT NOT NULL,
    yInicio INT NOT NULL,
    danio INT DEFAULT 0,
    FOREIGN KEY (idPartida) REFERENCES partidas(idPartida)
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;
